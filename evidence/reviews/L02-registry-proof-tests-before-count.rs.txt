#![allow(clippy::unwrap_used)]
use agentic_l2::{
    ProvenRegistrySnapshot, RegistryAnchor, RegistryError, RegistryProfile, verify_registry_at_root,
};
use alloy_primitives::{B256, U256};
use serde_json::{Value, json};

fn fixtures() -> Vec<Value> {
    serde_json::from_str(include_str!("vectors/registry-anvil.json")).unwrap()
}
fn hash(v: &Value) -> B256 {
    v.as_str().unwrap().parse().unwrap()
}
fn raw(v: &Value) -> Vec<u8> {
    serde_json::to_vec(v).unwrap()
}
fn profile(v: &Value) -> RegistryProfile {
    RegistryProfile::from_json(&raw(v)).unwrap()
}
fn anchor(v: &Value) -> RegistryAnchor {
    RegistryAnchor {
        state_root: hash(&v["stateRoot"]),
        block_number: v["blockNumber"].as_u64().unwrap(),
        block_timestamp: v["blockTimestamp"].as_u64().unwrap(),
        checked_at: v["checkedAt"].as_u64().unwrap(),
        lease_expires_at: v["leaseExpiresAt"].as_u64().unwrap(),
    }
}
fn check(p: &RegistryProfile, v: &Value) -> Result<ProvenRegistrySnapshot, RegistryError> {
    verify_registry_at_root(
        p,
        anchor(v),
        v["epoch"].as_u64().unwrap(),
        &raw(&v["proof"]),
    )
}

#[test]
fn real_anvil_snapshot_and_member_authenticate_the_whole_gapped_active_set() {
    for f in fixtures() {
        let selected = profile(&f["profile"]);
        for state in ["current", "postChanged"] {
            let v = &f[state];
            let snapshot = check(&selected, v).unwrap();
            let e = &f["expected"];
            assert_eq!(snapshot.state_root(), hash(&v["stateRoot"]));
            assert_eq!(snapshot.epoch(), e["epoch"].as_u64().unwrap());
            assert_eq!(snapshot.domain(), hash(&e["domain"]));
            assert_eq!(snapshot.root(), hash(&e["root"]));
            assert_eq!(snapshot.seed(), hash(&e["seed"]));
            assert_eq!(snapshot.count(), 2);
            assert_eq!(snapshot.sealed_at(), e["sealedAt"].as_u64().unwrap());
            assert_eq!(
                snapshot.admission_until(),
                e["admissionUntil"].as_u64().unwrap()
            );
            assert_eq!(snapshot.valid_until(), e["validUntil"].as_u64().unwrap());
            assert_eq!(snapshot.max_obligation_seconds(), 600);
            let member = snapshot
                .verify_member(
                    &raw(&f["member"]),
                    hash(&f["nodeKey"]),
                    hash(&f["salt"]),
                    v["checkedAt"].as_u64().unwrap(),
                )
                .unwrap();
            assert_eq!(member.index(), 2);
            assert_eq!(
                member.ordinal(),
                1,
                "ordinal is not append-only index after an exit"
            );
            assert_eq!(
                member.owner().to_string().to_lowercase(),
                f["member"]["owner"].as_str().unwrap().to_lowercase()
            );
            assert_eq!(member.commitment(), hash(&f["member"]["commitment"]));
            assert_eq!(member.node_key(), hash(&f["nodeKey"]));
            let first = snapshot
                .verify_member(
                    &raw(&f["firstMember"]),
                    hash(&f["firstNodeKey"]),
                    hash(&f["firstSalt"]),
                    v["checkedAt"].as_u64().unwrap(),
                )
                .unwrap();
            assert_eq!(
                (first.index(), first.ordinal()),
                (1, 0),
                "right sibling count must not increase ordinal"
            );
            assert_eq!(
                first.owner().to_string().to_lowercase(),
                f["firstMember"]["owner"].as_str().unwrap().to_lowercase()
            );
            assert_ne!(first.owner(), member.owner());
            assert_eq!(first.node_key(), hash(&f["firstNodeKey"]));
            assert_eq!(first.commitment(), hash(&f["firstMember"]["commitment"]));
            assert_eq!(
                f["firstMember"]["siblings"][1]["count"], 1,
                "first member must have a nonempty right sibling"
            );
        }
        let v = &f["current"];
        let slots: Vec<U256> = v["slots"]
            .as_array()
            .unwrap()
            .iter()
            .map(|x| x.as_str().unwrap().parse().unwrap())
            .collect();
        assert_eq!(
            selected
                .snapshot_slots(v["epoch"].as_u64().unwrap())
                .as_slice(),
            slots
        );
        let mut reversed = v.clone();
        reversed["proof"]["storageProof"]
            .as_array_mut()
            .unwrap()
            .reverse();
        assert_eq!(
            check(&selected, &reversed).unwrap(),
            check(&selected, v).unwrap()
        );
    }
}

#[test]
fn selected_chain_deployment_and_all_configuration_fields_bind_the_snapshot() {
    let f = &fixtures()[0];
    for (field, value) in [
        ("chainId", json!("0x14a34")),
        ("registry", json!(format!("0x{}", "a1".repeat(20)))),
        ("genesis", json!(format!("0x{}", "a1".repeat(32)))),
        ("codeHash", json!(format!("0x{}", "a1".repeat(32)))),
        ("unitBondWei", json!("0x66")),
        ("epochSeconds", json!(61)),
        ("snapshotLeaseSeconds", json!(121)),
        ("maxObligationSeconds", json!(601)),
        ("futureBlockDelay", json!(4)),
    ] {
        let mut p = f["profile"].clone();
        p[field] = value;
        assert!(
            check(&profile(&p), &f["current"]).is_err(),
            "unbound {field}"
        );
    }
}

#[test]
fn absent_unseeded_and_orphaned_snapshots_never_authorize_new_placement() {
    for f in fixtures() {
        let p = profile(&f["profile"]);
        for state in ["before", "absent", "rolledBack"] {
            assert_eq!(
                check(&p, &f[state]),
                Err(RegistryError::MissingSnapshot),
                "{state}"
            );
        }
        assert_eq!(check(&p, &f["unseeded"]), Err(RegistryError::Unseeded));
        let mut orphan = f["current"].clone();
        orphan["stateRoot"] = f["rolledBack"]["stateRoot"].clone();
        assert!(check(&p, &orphan).is_err());
        let mut wrong_epoch = f["current"].clone();
        wrong_epoch["epoch"] = f["absent"]["epoch"].clone();
        assert!(check(&p, &wrong_epoch).is_err());
        // Valid historical root membership alone still does not authenticate that root.
        assert!(check(&p, &f["current"]).is_ok());
    }
}

#[test]
fn account_fields_storage_values_and_every_mpt_path_are_verified() {
    let f = &fixtures()[0];
    let p = profile(&f["profile"]);
    for (field, value) in [
        ("balance", json!("0x0")),
        ("nonce", json!("0x2")),
        ("address", json!(format!("0x{}", "b2".repeat(20)))),
        ("codeHash", json!(format!("0x{}", "b2".repeat(32)))),
        ("storageHash", json!(format!("0x{}", "b2".repeat(32)))),
    ] {
        let mut v = f["current"].clone();
        assert_ne!(v["proof"][field], value);
        v["proof"][field] = value;
        assert!(check(&p, &v).is_err(), "unproven {field}");
    }
    for i in 0..5 {
        let mut v = f["current"].clone();
        v["proof"]["storageProof"][i]["value"] = json!("0x1");
        assert!(check(&p, &v).is_err(), "unproven word {i}");
    }
    let mut paths = vec!["/proof/accountProof".to_string()];
    paths.extend((0..5).map(|i| format!("/proof/storageProof/{i}/proof")));
    for path in paths {
        let nodes = f["current"].pointer(&path).unwrap().as_array().unwrap();
        assert!(!nodes.is_empty());
        for (i, node) in nodes.iter().enumerate() {
            let mut v = f["current"].clone();
            let mut damaged = node.as_str().unwrap().to_string();
            let end = damaged.len();
            let byte = if &damaged[end - 2..] == "00" {
                "01"
            } else {
                "00"
            };
            damaged.replace_range(end - 2.., byte);
            v.pointer_mut(&path).unwrap()[i] = json!(damaged);
            assert!(check(&p, &v).is_err(), "corrupted {path}/{i}");
        }
        let mut v = f["current"].clone();
        v.pointer_mut(&path).unwrap().as_array_mut().unwrap().pop();
        assert!(check(&p, &v).is_err(), "truncated {path}");
    }
    let mut v = f["current"].clone();
    v["proof"]["storageProof"][1]["key"] = v["proof"]["storageProof"][0]["key"].clone();
    assert!(check(&p, &v).is_err());
    let mut v = f["current"].clone();
    v["proof"]["storageProof"].as_array_mut().unwrap().pop();
    assert!(check(&p, &v).is_err());
    let mut v = f["current"].clone();
    let extra = v["proof"]["storageProof"][0].clone();
    v["proof"]["storageProof"]
        .as_array_mut()
        .unwrap()
        .push(extra);
    assert!(check(&p, &v).is_err());
}

#[test]
fn valid_mpt_cannot_hide_invalid_registry_state_or_finite_horizons() {
    let f = &fixtures()[0];
    let p = profile(&f["profile"]);
    assert_eq!(f["invalidStates"].as_array().unwrap().len(), 9);
    for v in f["invalidStates"].as_array().unwrap() {
        assert_eq!(v["injectedStorage"], true);
        let expected = match v["case"].as_str().unwrap() {
            "wrong_domain" => RegistryError::Domain,
            "zero_seed" => RegistryError::Unseeded,
            "future_seal" => RegistryError::Time,
            "zero_count" | "excess_count" | "zero_root" | "reserved_bits" | "wrong_admission"
            | "wrong_beacon" => RegistryError::Snapshot,
            other => panic!("unknown injected state {other}"),
        };
        assert_eq!(check(&p, v), Err(expected), "{}", v["case"]);
    }
}

#[test]
fn snapshot_and_cached_membership_expire_at_the_earlier_checkpoint_or_admission_bound() {
    let f = &fixtures()[0];
    let p = profile(&f["profile"]);
    let current = &f["current"];
    let now = current["checkedAt"].as_u64().unwrap();
    let admission = f["expected"]["admissionUntil"].as_u64().unwrap();
    for until in [now + 30, admission] {
        let mut v = current.clone();
        v["leaseExpiresAt"] = json!(until);
        let s = check(&p, &v).unwrap();
        assert_eq!(s.valid_until(), until);
        assert!(
            s.verify_member(
                &raw(&f["member"]),
                hash(&f["nodeKey"]),
                hash(&f["salt"]),
                until - 1
            )
            .is_ok()
        );
        for denied in [now - 1, until, until + 1] {
            assert_eq!(
                s.verify_member(
                    &raw(&f["member"]),
                    hash(&f["nodeKey"]),
                    hash(&f["salt"]),
                    denied
                ),
                Err(RegistryError::Time)
            );
        }
        v["checkedAt"] = json!(until - 1);
        assert!(check(&p, &v).is_ok());
        v["checkedAt"] = json!(until);
        assert_eq!(check(&p, &v), Err(RegistryError::Time));
    }
    let mut v = current.clone();
    v["checkedAt"] = json!(now - 1);
    assert_eq!(check(&p, &v), Err(RegistryError::Time));
    let mut v = current.clone();
    v["blockNumber"] = json!(current["blockNumber"].as_u64().unwrap() - 1);
    assert!(check(&p, &v).is_err(), "seed used before capture block");
}

#[test]
fn operator_opening_owner_order_and_full_subtree_counts_are_not_peer_assertions() {
    let f = &fixtures()[0];
    let s = check(&profile(&f["profile"]), &f["current"]).unwrap();
    let now = f["current"]["checkedAt"].as_u64().unwrap();
    let verify = |v: &Value, key: B256, salt: B256| s.verify_member(&raw(v), key, salt, now);
    let key = hash(&f["nodeKey"]);
    let salt = hash(&f["salt"]);
    for (k, t) in [
        (B256::ZERO, salt),
        (B256::repeat_byte(0xa1), salt),
        (key, B256::repeat_byte(0xa1)),
    ] {
        assert!(verify(&f["member"], k, t).is_err());
    }
    for (field, value) in [
        ("index", json!(1)),
        ("owner", json!(format!("0x{}", "b2".repeat(20)))),
        ("commitment", json!(format!("0x{}", "b2".repeat(32)))),
    ] {
        let mut v = f["member"].clone();
        v[field] = value;
        assert!(verify(&v, key, salt).is_err(), "accepted wrong {field}");
    }
    for level in 0..32 {
        for field in ["hash", "count"] {
            let mut v = f["member"].clone();
            v["siblings"][level][field] = if field == "hash" {
                json!(format!("0x{}", "b2".repeat(32)))
            } else {
                json!(v["siblings"][level][field].as_u64().unwrap() + 1)
            };
            assert!(
                verify(&v, key, salt).is_err(),
                "unproven sibling {level}/{field}"
            );
        }
    }
    let mut v = f["member"].clone();
    v["siblings"].as_array_mut().unwrap().swap(0, 1);
    assert!(verify(&v, key, salt).is_err());
    let mut v = f["member"].clone();
    v["siblings"].as_array_mut().unwrap().pop();
    assert!(verify(&v, key, salt).is_err());
    let mut v = f["member"].clone();
    let extra = v["siblings"][0].clone();
    v["siblings"].as_array_mut().unwrap().push(extra);
    assert!(verify(&v, key, salt).is_err());
    let mut v = f["member"].clone();
    v["operatorIndependence"] = json!(true);
    assert!(verify(&v, key, salt).is_err());
}

#[test]
fn registry_profiles_and_untrusted_proof_schemas_are_bounded_and_strict() {
    let f = &fixtures()[0];
    let p = profile(&f["profile"]);
    let current = &f["current"];
    let verify = |bytes: &[u8]| {
        verify_registry_at_root(
            &p,
            anchor(current),
            current["epoch"].as_u64().unwrap(),
            bytes,
        )
    };
    assert_eq!(
        RegistryProfile::from_json(&vec![b' '; 16385]).unwrap_err(),
        RegistryError::Limit
    );
    assert_eq!(
        verify(&vec![b' '; 1024 * 1024 + 1]),
        Err(RegistryError::Limit)
    );
    for (field, value) in [
        ("chainId", json!("0x0")),
        ("unitBondWei", json!("0x0")),
        ("epochSeconds", json!(59)),
        ("epochSeconds", json!(86401)),
        ("snapshotLeaseSeconds", json!(59)),
        ("snapshotLeaseSeconds", json!(604801)),
        ("maxObligationSeconds", json!(0)),
        ("maxObligationSeconds", json!(31622401)),
        ("futureBlockDelay", json!(0)),
        ("futureBlockDelay", json!(129)),
    ] {
        let mut v = f["profile"].clone();
        v[field] = value;
        assert!(RegistryProfile::from_json(&raw(&v)).is_err(), "{field}");
    }
    let mut duplicate = serde_json::to_string(&f["profile"]).unwrap();
    duplicate.insert_str(1, "\"chainId\":\"0x1\",");
    assert_eq!(
        serde_json::from_str::<Value>(&duplicate).unwrap(),
        f["profile"]
    );
    assert!(RegistryProfile::from_json(duplicate.as_bytes()).is_err());
    let mut duplicate = serde_json::to_string(&current["proof"]).unwrap();
    duplicate.insert_str(1, "\"balance\":\"0x0\",");
    assert_eq!(
        serde_json::from_str::<Value>(&duplicate).unwrap(),
        current["proof"]
    );
    assert!(verify(duplicate.as_bytes()).is_err());
    let mut v = current["proof"].clone();
    v["finalized"] = json!(true);
    assert!(verify(&raw(&v)).is_err());
    let mut v = current["proof"].clone();
    let node = v["accountProof"][0].clone();
    v["accountProof"] = json!(vec![node; 65]);
    assert!(verify(&raw(&v)).is_err());
    let mut v = current["proof"].clone();
    v["accountProof"][0] = json!(format!("0x{}", "00".repeat(1025)));
    assert!(verify(&raw(&v)).is_err());
    let s = check(&p, current).unwrap();
    let now = current["checkedAt"].as_u64().unwrap();
    assert_eq!(
        s.verify_member(
            &vec![b' '; 16385],
            hash(&f["nodeKey"]),
            hash(&f["salt"]),
            now
        ),
        Err(RegistryError::Limit)
    );
    let mut duplicate = serde_json::to_string(&f["member"]).unwrap();
    duplicate.insert_str(1, "\"index\":0,");
    assert_eq!(
        serde_json::from_str::<Value>(&duplicate).unwrap(),
        f["member"]
    );
    assert!(
        s.verify_member(
            duplicate.as_bytes(),
            hash(&f["nodeKey"]),
            hash(&f["salt"]),
            now
        )
        .is_err()
    );
}
